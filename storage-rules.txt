////////////////////////////////
// STORAGE RULES (DEV & PROD)  //
////////////////////////////////

////////////////////////////////
// DEV RULES (pengembangan)    //
////////////////////////////////
// - Tujuan: cepat uji CRUD upload Storage
// - Baca publik untuk semua file (agar gambar tampil di publik)
// - Tulis diizinkan untuk user yang sudah login (authenticated)
// - Gunakan hanya saat development, jangan untuk produksi
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Public read (agar gambar bisa di-load)
    match /{allPaths=**} {
      allow read: if true;
    }

    // Write untuk user login (DEV)
    match /images/{allPaths=**} {
      allow write: if request.auth != null;
    }
  }
}


////////////////////////////////
// PROD RULES (disarankan)    //
////////////////////////////////
// - Tujuan: aman untuk produksi
// - Baca publik untuk semua file
// - Tulis hanya untuk admin (custom claim admin: true)
// - Setelah set admin claim, minta admin logout/login agar token refresh
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Public read (agar gambar bisa di-load oleh website publik)
    match /{allPaths=**} {
      allow read: if true;
    }

    // Hanya admin boleh upload/ubah/hapus file di dalam folder images/
    match /images/{allPaths=**} {
      allow write: if request.auth != null && request.auth.token.admin == true;
    }
  }
}
