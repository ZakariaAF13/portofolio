rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Temporary DEV rules - allow authenticated users to write
    match /{collectionId}/{documentId} {
      allow read: if true;
      allow write: if request.auth != null && collectionId in [
        'portfolio_projects',
        'portfolio_skills', 
        'portfolio_profile',
        'portfolio_settings',
        'portfolio_what_i_do',
        'portfolio_knowledge',
        'portfolio_experiences',
        'portfolio_educations',
        'portfolio_sections'
      ];
    }
  }
}
